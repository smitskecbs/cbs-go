// Local leaderboard v1 (no backend yet) // Reads XP/Level directly from localStorage so it won't break when state.js changes. const LB_KEY = 'cbsgo_leaderboard_v1'; const NAME_KEY = 'cbsgo_player_name_v1'; const STATE_KEY = 'cbsgo_state_v1'; function loadJson(key, fallback) { try { const raw = localStorage.getItem(key); if (!raw) return fallback; const data = JSON.parse(raw); return data ?? fallback; } catch { return fallback; } } function saveJson(key, value) { try { localStorage.setItem(key, JSON.stringify(value)); } catch { // ignore } } function randomName() { const a = ['Sovereign','Builder','Runner','Scout','Mapper','Guardian','Wanderer','Cipher']; const b = Math.random().toString(16).slice(2, 6).toUpperCase(); return `${a[Math.floor(Math.random() * a.length)]}-${b}`; } export function getPlayerName() { let name = ''; try { name = localStorage.getItem(NAME_KEY) || ''; } catch {} if (!name) { name = randomName(); try { localStorage.setItem(NAME_KEY, name); } catch {} } return name; } export function setPlayerName(name) { const clean = String(name || '').trim().slice(0, 24); if (!clean) return getPlayerName(); try { localStorage.setItem(NAME_KEY, clean); } catch {} return clean; } function readXpLevel() { const s = loadJson(STATE_KEY, {}); const xp = Number(s?.xp ?? 0); const level = Number(s?.level ?? 1); return { xp: Number.isFinite(xp) ? xp : 0, level: Number.isFinite(level) ? level : 1 }; } export function getTopScores(limit = 10) { const list = loadJson(LB_KEY, []); const arr = Array.isArray(list) ? list : []; arr.sort((x, y) => Number(y?.xp ?? 0) - Number(x?.xp ?? 0)); return arr.slice(0, limit); } export function submitMyScore() { const { xp, level } = readXpLevel(); const name = getPlayerName(); const entry = { name, xp, level, ts: Date.now() }; const list = loadJson(LB_KEY, []); const arr = Array.isArray(list) ? list : []; const idx = arr.findIndex(e => e?.name === name); if (idx >= 0) { // keep best XP if (entry.xp >= Number(arr[idx]?.xp ?? 0)) arr[idx] = entry; } else { arr.push(entry); } arr.sort((x, y) => Number(y?.xp ?? 0) - Number(x?.xp ?? 0)); saveJson(LB_KEY, arr.slice(0, 50)); return entry; }
